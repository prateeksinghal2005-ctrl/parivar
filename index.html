// ============================================================ // Yahan se neeche kuch badalne ki zaroorat nahi // ============================================================ var path = [family]; // photo ka path GitHub ke hisaab se theek karta hai function avatar(p, cls) { if (p.photo) { var i = document.createElement("img"); i.src = p.photo; i.alt = p.name; i.className = cls + " zoom"; i.onclick = function (e) { e.stopPropagation(); openPhoto(p); }; return i; } var d = document.createElement("div"); d.className = cls + " blank"; return d; } function openPhoto(p) { document.getElementById("bigphoto").src = p.photo; document.getElementById("cap").textContent = p.name + (p.dob ? " | " + p.dob : ""); document.getElementById("overlay").style.display = "flex"; } document.getElementById("overlay").onclick = function () { this.style.display = "none"; }; document.addEventListener("keydown", function (e) { if (e.key === "Escape") document.getElementById("overlay").style.display = "none"; }); function bigCard(p) { var d = document.createElement("div"); d.className = "pcard"; d.appendChild(avatar(p, "big")); var b = document.createElement("b"); b.textContent = p.name; d.appendChild(b); if (p.dob) { var s = document.createElement("small"); s.textContent = p.dob; d.appendChild(s); } return d; } function show() { var person = path[path.length - 1]; document.getElementById("path").textContent = path.map(function (p) { return p.name; }).join(" > "); var people = document.getElementById("people"); people.innerHTML = ""; people.appendChild(bigCard(person)); if (person.spouse) people.appendChild(bigCard(person.spouse)); document.getElementById("back").style.display = path.length > 1 ? "inline-block" : "none"; var list = document.getElementById("list"); list.innerHTML = ""; if (person.children.length === 0) { document.getElementById("title").textContent = "Koi bachcha nahi hai"; return; } document.getElementById("title").textContent = person.name + " ke bachche:"; person.children.forEach(function (child) { var li = document.createElement("li"); li.appendChild(avatar(child, "small")); var box = document.createElement("div"); var b = document.createElement("b"); b.textContent = child.name; box.appendChild(b); if (child.dob) { var s = document.createElement("small"); s.textContent = child.dob; box.appendChild(s); } if (child.spouse) { var sp = document.createElement("div"); sp.className = "sp"; sp.appendChild(avatar(child.spouse, "tiny")); var t = document.createElement("span"); t.textContent = child.spouse.name; sp.appendChild(t); box.appendChild(sp); } li.appendChild(box); li.onclick = function () { path.push(child); show(); }; list.appendChild(li); }); } document.getElementById("back").onclick = function () { if (path.length > 1) { path.pop(); show(); } }; show();
